--@include ./wgui.lua


local function setup()
    enableHud( player(), true )
    require( "./wgui.lua" )
    wgui.debug = false
    
    local e = {}
    
    e.rs = wgui.createRenderSpace( RENDERSPACE.SCREEN )
    e.rs:setScreen( chip():getLinkedComponents()[ 1 ] )
    
    e.space = wgui.create( "space", e.rs )
    e.space:setPos( 64, 64 )
    e.space:setSize( 1024 - 128, 1024 - 128 )
    
    local btn1 = wgui.create( "button", e.space )
    btn1:setPos( 32, 32 )
    btn1:setSize( 32, 32 )
    
    local btn2
    
    for xx = 1, 10 do
        btn2 = wgui.create( "button", e.space )
        btn2:setPos( 256, 64 + xx * 48 )
        btn2:setSize( 32, 32 )
        btn2.linkable = true
    end
    
    -- bezier shit
    btn1.prevpaint = btn1.paint
    
    btn1.paint = function( self )
        if btn1.bzra or btn1.link then
            btn1.bzr.v1 = Vector( btn1.data.positionGlobal.x + btn1.data.sizeGlobal.w / 2, btn1.data.positionGlobal.y + btn1.data.sizeGlobal.h / 2 )
            
            if btn1.bzra then
                btn1.bzr.v4 = Vector( btn1.data.renderSpace.cursor.position.x, btn1.data.renderSpace.cursor.position.y )
            else
                btn1.bzr.v4 = Vector( btn1.link.data.positionGlobal.x + btn1.link.data.sizeGlobal.w / 2, btn1.link.data.positionGlobal.y + btn1.link.data.sizeGlobal.h / 2 )
            end
            
            local dist = ( btn1.bzr.v4.x - ( btn1.data.positionGlobal.x + btn1.data.sizeGlobal.w / 2 ) ) * 0.66
            
            btn1.bzr.v2 = btn1.bzr.v1 + Vector( dist, 0 )
            btn1.bzr.v3 = btn1.bzr.v4 + Vector( -dist, 0 )
            
            local step = 10
            
            for iter = 0, 100, step do
                if iter == 0 then continue end
                
                local ppp = math.bezierVectorCubic( 0.01 * ( iter - step ), btn1.bzr.v1, btn1.bzr.v2, btn1.bzr.v3, btn1.bzr.v4 )
                local nnn = math.bezierVectorCubic( 0.01 * iter, btn1.bzr.v1, btn1.bzr.v2, btn1.bzr.v3, btn1.bzr.v4 )
                
                render.setRGBA( 255, 0, 0, 255 )
                render.drawLine( ppp.x, ppp.y, nnn.x, nnn.y )
            end
        end
        
        self:prevpaint()
    end
    
    btn1.bzr = { v1 = Vector(), v2 = Vector(), v3 = Vector(), v4 = Vector() }
    btn1.bzra = false
    btn1.link = btn2
    
    btn1:addEvent( "click", function( self )
        self.bzra = true
    end )
    
    btn1:addEvent( "clickrelease", function( self )
        self.bzra = false
        local hover = self.data.renderSpace.data.hoverElement
        self.link = hover.linkable == true and hover or nil
    end )
    
    
    
    local textbox = wgui.create( "textbox", e.space )
    textbox:setPos( 512, 60 )
    textbox:setSize( 256, 58 )
    
    textbox:addEvent( "valuechanged", function( self, val, oval )
        print( val )
    end )
    
    local textbox = wgui.create( "textbox", e.space )
    textbox:setPos( 512, 120 )
    textbox:setSize( 256, 58 )
    
    textbox:addEvent( "valuechanged", function( self, val, oval )
        print( val )
    end )
    
    local textbox = wgui.create( "textbox", e.space )
    textbox:setPos( 512, 180 )
    textbox:setSize( 256, 58 )
    
    textbox:addEvent( "valuechanged", function( self, val, oval )
        print( val )
    end )
end


hook.add( "PermissionRequest", "PermissionRequest11", function()
    if permissionRequestSatisfied() then setup() end
end )

local rss = {}

    card:setPos( 64, 64 )
    card:setSize( 512 - 128, 512 - 128 )
    
    card.matrix = Matrix()
    
    local mat = render.createMaterial( 
        "https://dl.dropbox.com/scl/fi/nshqjt2kq8f24y58vyb06/a-37.jpg?rlkey=fo28otawvmrfqgybd0qkx8rdx&st=e6i2g6vo", 
        function( _, _, _, _, layout ) if layout then layout( 0, 0, 1024, 1024 ) end end
    )
    
    card.paint = function( self )
        render.setRGBA( 255, 255, 255, 255 )
        render.setMaterial(mat)
        render.drawTexturedRect( self.data.positionGlobal.x, self.data.positionGlobal.y, self.data.sizeGlobal.w, self.data.sizeGlobal.h )
    end
    
    card.newpaint = function( self )
        local crs = self.data.renderSpace.cursor.position
        local angx = ( crs.x - ( self.data.positionGlobal.x + self.data.sizeGlobal.w / 2 ) ) / ( self.data.sizeGlobal.w / 2 )
        local angy = ( crs.y - ( self.data.positionGlobal.y + self.data.sizeGlobal.h / 2 ) ) / ( self.data.sizeGlobal.h / 2 )
        
        render.pushViewMatrix( {
            type="3D",
            x = self.data.positionGlobal.x - 0,
            y = self.data.positionGlobal.y - 0,
            w = self.data.sizeGlobal.w + 0,
            h = self.data.sizeGlobal.h + 0,
            fov = 100,
            origin = Vector( -200, 0, 0 ), -- -79
            angles = Angle( 0, 0, 90 ),
            aspect = 1
        } )
            
            self.matrix:setAngles( Angle( 90 + -angx * 7.5, -angy * 7.5, 0 ) )
            
            render.pushMatrix( self.matrix, false )
                render.setRGBA( 255, 255, 255, 255 )
                render.setMaterial(mat)
                render.drawTexturedRect( -self.data.sizeGlobal.w / 2, -self.data.sizeGlobal.h / 2, self.data.sizeGlobal.w, self.data.sizeGlobal.h )
                
                local fade = angy < 0 and 255 or 0
                render.setRGBA( fade, fade, fade, 100 * math.abs( angy ) )
                render.drawRectFast( -self.data.sizeGlobal.w / 2, -self.data.sizeGlobal.h / 2, self.data.sizeGlobal.w, self.data.sizeGlobal.h )
            render.popMatrix()
        
        render.popViewMatrix()
    end
    
    card:addEvent( "hoveron", function( self )
        self.oldpaint = self.paint
        self.paint = self.newpaint
    end )
    
    card:addEvent( "hoveroff", function( self )
        self.paint = self.oldpaint
    end )
end
